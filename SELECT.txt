SELECT
	p.nome AS nome_produto,
	CASE
        WHEN pp.preco_promocional_varejo <> 0 THEN pp.preco_promocional_varejo
        WHEN pp.preco_promocional_atacado <> 0 THEN pp.preco_promocional_atacado
        WHEN pp.percentual_promocional_varejo <> 0 THEN p.preco_venda - (p.preco_venda * (pp.percentual_promocional_varejo / 100))
        WHEN pp.percentual_promocional_atacado <> 0 THEN p.preco_venda - (p.preco_venda * (pp.percentual_promocional_atacado / 100))
        ELSE p.preco_venda
    END AS preco_atual,
    pp.id_produto AS id,
    p.preco_venda AS preco_anterior,
	prodi.caminho_imagem AS link_imagem,
	null as data_validade,
	promo.data_inicial as data_inicio,
	promo.data_final as data_fim,
	'3' as dias_semana,
	'08:00' as hora_inicio,
	'17:05' as hora_fim,
	null AS texto_validade
FROM 
    promocao_produto pp
    LEFT JOIN produto p ON p.id_produto = pp.id_produto
	LEFT JOIN promocao promo on promo.id_promocao = pp.id_promocao
    LEFT JOIN (
        SELECT 
            id_produto,
            caminho_imagem,
            ROW_NUMBER() OVER (
                PARTITION BY id_produto 
                ORDER BY imagem_principal DESC
            ) AS rn
        FROM produto_imagem
    ) prodi ON prodi.id_produto = p.id_produto AND prodi.rn = 1

	/*
	0 = Domingo
	1 = Segunda-feira
	2 = Terça-feira
	3 = Quarta-feira
	4 = Quinta-feira
	5 = Sexta-feira
	6 = Sábado
	*/
